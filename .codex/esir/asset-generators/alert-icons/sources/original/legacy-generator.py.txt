class Glyph:
    def __init__(self, color):
        self.color = COLORS[color]
        self.commands = []
    def line(self, points, color=None, width=7):
        self.commands.append(('line', points, color or self.color, width))
    def poly(self, points, fill=None, stroke=METAL, width=3):
        self.commands.append(('polygon', points, fill or self.color, stroke, width))
    def ellipse(self, box, fill=None, stroke=METAL, width=3):
        self.commands.append(('ellipse', box, fill or self.color, stroke, width))
    def arc(self, center, radius, start, end, color=None, width=7):
        self.line([(center[0]+cos(t*pi/180)*radius,center[1]+sin(t*pi/180)*radius)
                   for t in range(start,end+1,4)], color, width)
    def leaf(self, x=64, y=62, size=32, wilt=False):
        self.poly([(x-size,y+size//2),(x-size*.7,y-size*.4),(x+size*.15,y-size),
                   (x+size,y-size*.8),(x+size*.8,y+size*.2),(x,y+size*.7)])
        self.line([(x-size*.7,y+size*.6),(x+size*.6,y-size*.5)],INK,5)
        if wilt:self.line([(x-18,y+21),(x-15,y+34),(x+1,y+40)],METAL,5)
    def sprout(self,x=64,y=80):
        self.line([(x,y+14),(x,y-17)],METAL,6)
        self.poly([(x,y-5),(x-24,y-8),(x-28,y-28),(x-8,y-23)])
        self.poly([(x,y-12),(x+8,y-31),(x+29,y-32),(x+24,y-13)])
    def ground(self,y=94):
        self.line([(30,y),(98,y)],METAL,6)
    def crack(self,x=64,y=50):
        self.line([(x,y),(x-8,y+15),(x+5,y+22),(x-7,y+40)],INK,6)
    def drop(self,x=64,y=62,s=28):
        self.poly([(x,y-s),(x+s*.75,y+s*.25),(x+s*.65,y+s*.75),(x,y+s),
                   (x-s*.65,y+s*.75),(x-s*.75,y+s*.25)])
    def pipe(self):
        self.poly([(26,63),(54,63),(48,79),(56,94),(26,94)])
        self.poly([(77,63),(102,63),(102,94),(75,94),(81,79)])
        self.line([(29,58),(29,99)],METAL,7);self.line([(99,58),(99,99)],METAL,7)
    def flame(self,x=64,y=58,s=30):
        self.poly([(x,y-s),(x+6,y-5),(x+18,y-18),(x+s*.8,y+14),(x+12,y+s),
                   (x-14,y+s),(x-s*.8,y+12),(x-15,y-8),(x-10,y+5)])
    def tree(self):
        self.poly([(64,25),(87,54),(76,54),(101,81),(69,81),(69,102),(59,102),(59,81),(27,81),(51,54),(40,54)])

def render(g):
    image=Image.new('RGBA',(128*SCALE,128*SCALE));draw=ImageDraw.Draw(image)
    body=[(28,8),(100,8),(120,28),(120,100),(100,120),(28,120),(8,100),(8,28)]
    commands=[('polygon',body,'#252d32','#807a67',3)]+g.commands
    svg=['<svg xmlns="http://www.w3.org/2000/svg" width="128" height="128" viewBox="0 0 128 128">']
    for cmd in commands:
        kind,coords,*args=cmd
        if kind=='line':
            color,width=args;points=[(round(x*SCALE),round(y*SCALE)) for x,y in coords]
            draw.line(points,fill=color,width=width*SCALE,joint='curve')
            for x,y in (points[0],points[-1]):draw.ellipse((x-width*SCALE/2,y-width*SCALE/2,x+width*SCALE/2,y+width*SCALE/2),fill=color)
            svg.append(f'<polyline points="{" ".join(f"{x:g},{y:g}" for x,y in coords)}" fill="none" stroke="{color}" stroke-width="{width}" stroke-linecap="round" stroke-linejoin="round"/>')
        elif kind=='polygon':
            fill,stroke,width=args;points=[(round(x*SCALE),round(y*SCALE)) for x,y in coords]
            draw.polygon(points,fill=fill);draw.line(points+[points[0]],fill=stroke,width=width*SCALE,joint='curve')
            svg.append(f'<polygon points="{" ".join(f"{x:g},{y:g}" for x,y in coords)}" fill="{fill}" stroke="{stroke}" stroke-width="{width}" stroke-linejoin="round"/>')
        else:
            fill,stroke,width=args;x1,y1,x2,y2=coords
            draw.ellipse(tuple(round(n*SCALE) for n in coords),fill=fill,outline=stroke,width=width*SCALE)
            svg.append(f'<ellipse cx="{(x1+x2)/2:g}" cy="{(y1+y2)/2:g}" rx="{(x2-x1)/2:g}" ry="{(y2-y1)/2:g}" fill="{fill}" stroke="{stroke}" stroke-width="{width}"/>')
    svg.append('</svg>')
    return image.resize((128,128),Image.Resampling.LANCZOS),'\n'.join(svg)+'\n'

def main():
    parser=argparse.ArgumentParser();parser.add_argument('--preview-dir',type=Path,required=True);args=parser.parse_args()
    target=ROOT/'exotic-space-industries-remembrance/graphics/gui/alerts';target.mkdir(parents=True,exist_ok=True)
    sources=HERE/'sources';sources.mkdir(exist_ok=True);args.preview_dir.mkdir(parents=True,exist_ok=True)
    icons=make_icons();assert len(icons)==27
    sheet=Image.new('RGB',(1080,9*170),'#25292e');draw=ImageDraw.Draw(sheet)
    for index,(name,glyph) in enumerate(icons.items()):
        bitmap,svg=render(glyph);bitmap.save(target/(name+'.png'),optimize=True);(sources/(name+'.svg')).write_text(svg,encoding='utf-8')
        x=(index%3)*360;y=(index//3)*170;sheet.paste(bitmap,(x+8,y+6),bitmap)
        for offset,size in enumerate((16,24,32,64)):
            small=bitmap.resize((size,size),Image.Resampling.LANCZOS);sheet.paste(small,(x+148+offset*46,y+20),small)
        draw.text((x+8,y+140),name,fill='#eee9db')
    sheet.save(args.preview_dir/'contact-sheet.png');print(f'Exported {len(icons)} main-mod PNGs and retained SVG sources')

if __name__=='__main__':main()
